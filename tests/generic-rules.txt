# Generic rules used by tests/engine.test.js (the simple ALGORITHM language, not the CDF algorithm)
ALGORITHM "CDF status update from INTEGRATOR"
BASE FIRST
UPDATE "INTEGRATOR"
BASE_START_ROW 3
UPDATE_START_ROW 2
MATCH UPDATE.A = BASE.C
SKIP_EMPTY NO
DATE_FORMAT "dd/mm/yyyy"
COPY UPDATE.M -> BASE.T
IF BASE.T = "9 - FINAL APPROVAL" COPY UPDATE.N -> BASE.X
IF BASE.T = "9 - FINAL APPROVAL" COPY UPDATE.Q -> BASE.AA
IF BASE.T = "9 - FINAL APPROVAL" COPY UPDATE.R -> BASE.AB
ON_CHANGE SET BASE.R = TODAY
REPORT "CDFs in FINAL APPROVAL" WHERE BASE.T = "9 - FINAL APPROVAL" COLUMNS C, T, X
LOG "UPDATE LOG"
