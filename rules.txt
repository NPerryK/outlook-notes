# ---------------- Step 1: import files ----------------
# IMPORT <number> "<sheet in this workbook>"  [FILE "<file name>"]  [FROM "<sheet in the imported file>"]
# The sheet in this workbook is deleted and fully replaced by the imported sheet (same position).
# FILE  - the file that belongs in this slot. Files dropped anywhere in step 1 go to the slot whose
#         name they contain (case, spaces and _ are ignored), and a different file gets a warning.
# FROM  - which sheet to take from the imported file. Without FROM, its first sheet is used.
# If the sheet doesn't exist in this workbook yet, it is created at the end.

# ---------------- New CDF row (from a PDF pair) ----------------
# Claude fills the fields from the CDF PDF and the ATT PDF (via claude.ai, copy-paste),
# and you review the form and approve. Where each CDF goes is explained below (NEWROW_MATCH).
#
# Possible sources for each column:
#   CDF              the CDF number without ".-" (A330-1-973)
#   CDF FULL         the CDF number as printed (A330-1-973.- / A330-1-682.A)
#   CDF REV          the CDF revision: "-" or the letter (A, B ...)
#   TODAY / NOW      today's date / date and time
#   "text"           a fixed value
#   AI.<FIELD>       DATE  PRIORITY  DESCRIPTION  OEM  DRAWING  INSTALLATION  CODE  DWG_REV  AUTHOR  ADVISER
#   LINK MAIN "<folder>"   link to the CDF PDF (folder + file name)
#   LINK ATT  "<folder>"   link to the ATT PDF
#
# The pane shows, next to every field, the header of the column it will be written to.

NEWROW_SHEET "CDF"
NEWROW_KEY C
NEWROW_HEADER_ROW 1
NEWROW_START_ROW 2

# Where a new CDF goes:
#  - already in column C            -> that row is updated
#  - a prepared row (column B already holds the CDF number and the PDF links, C is empty)
#                                   -> that row (NEWROW_MATCH = the column of the prepared numbers)
#  - a revision (.A, .B) of a CDF   -> a new row inserted right after the original CDF's block
#  - none of these                  -> a new row after the last used row
NEWROW_MATCH B

# Columns follow the header row of sheet CDF. A and B (links to the PDFs) are already in the
# prepared rows, so the add-in doesn't touch them. To have it write links, add for example:
#   NEWROW A = LINK ATT "<folder of the PDFs>\"
#   NEWROW B = LINK MAIN "<folder of the PDFs>\"
NEWROW C = CDF FULL           # LINK TO GCC        - the CDF number as printed, e.g. A330-1-973.-
NEWROW D = CDF REV            # CDF REVISION       - "-", or the revision letter (A, B ...)
NEWROW F = TODAY              # RECEIVED DATE
NEWROW G = AI.PRIORITY        # PRIORITY
NEWROW H = AI.DESCRIPTION     # PROBLEM DESCRIPTION
NEWROW I = AI.OEM             # TAGS
NEWROW J = AI.DRAWING         # DWG.
NEWROW K = AI.INSTALLATION    # DWG. Name
NEWROW L = AI.CODE            # NRC
NEWROW M = AI.DWG_REV         # CDF DWG REV
NEWROW N = AI.AUTHOR          # AUTHOR
NEWROW O = AI.ADVISER         # TECHNICAL ADVISER
